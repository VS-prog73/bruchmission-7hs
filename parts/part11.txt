function fracHTML(s){
  s = String(s);
  // Replace simple fraction tokens a/b with stacked spans, preserving units and operations.
  return s.replace(/(\d+)\/(\d+)/g, (_,a,b)=>`<span class="frac"><span class="top">${a}</span><span class="bottom">${b}</span></span>`)
          .replace(/\*/g,'<span class="op">·</span>')
          .replace(/:/g,'<span class="op">:</span>');
}
