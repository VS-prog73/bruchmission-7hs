b.disabled=locked);
  $('hintBtn').disabled = locked;
}
function startRetryDelay(){
  clearRetryDelay();
  waitLocked=true;
  let remaining = RETRY_DELAY_SECONDS;
  setAnswerLock(true);
  const updateCountdown = () => {
    $('feedback').className='feedback bad';
    $('feedback').innerHTML = `
      <div><b>Überlege noch einmal.</b> Das war noch nicht die passende Lösung.</div>
      <div class="countdown-box">
        <span class="countdown-number">${remaining}</span>
        <span><span class="countdown-text">Team-Denkzeit läuft.</span><br>
        <span class="countdown-small">Sprecht den Rechenweg noch einmal gemeinsam: Was ist gegeben? Welche Rechenart passt?</span></span>
      </div>`;
  };
  updateCountdown();
  waitTimer = setInterval(() => {
    remaining--;
    if(remaining <= 0){
      clearRetryDelay();
      setAnswerLock(false);
      $('feedback').className='feedback warn';
      $('feedback').innerHTML = `Jetzt dürft ihr neu wählen. <b>Tipp:</b> Erst rechnen, dann Antwort vergleichen.`;
    }else{
      updateCountdown();
    }
  }, 1000);
}

  if(t.includes('lagertor')||t.includes('trinkflaschen')||t.includes('feldküche')) return '🦊 Fino';
  return '🐯 Nala';
}
